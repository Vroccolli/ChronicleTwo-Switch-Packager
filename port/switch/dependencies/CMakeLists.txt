cmake_minimum_required(VERSION 3.25)
project(chronicletwo_switch_dependencies LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CT_PC_BUILD "${CT_ROOT}/port/build/windows-x64-release" CACHE PATH "Prepared PC dependencies")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch" AND NOT CT_SWITCH_PLATFORM)
    message(FATAL_ERROR "Use devkitPro/cmake/Switch.cmake; desktop libraries cannot be linked")
endif()
add_compile_options(-fsigned-char -ffp-contract=off -fPIE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory("${CT_PC_BUILD}/_deps/luau-src" luau EXCLUDE_FROM_ALL)
# GCC/newlib does not provide <climits> through the transitive headers used by
# this pinned VM's integer builtins. Supply the standard header explicitly.
target_compile_options(Luau.VM PRIVATE -include climits)
set(SIMDJSON_DEVELOPER_MODE OFF CACHE BOOL "" FORCE)
set(SIMDJSON_BUILD_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory("${CT_PC_BUILD}/_deps/simdjson-src" simdjson EXCLUDE_FROM_ALL)
# This simdjson version enables its optional mapped-file helper on every
# non-Windows OS. Horizon has no mmap; retain its buffered file/byte APIs.
set(CT_SIMDJSON_HEADERS "${CMAKE_BINARY_DIR}/simdjson-include")
file(COPY "${CT_PC_BUILD}/_deps/simdjson-src/include/" DESTINATION "${CT_SIMDJSON_HEADERS}")
foreach(header padded_string.h padded_string-inl.h)
    set(path "${CT_SIMDJSON_HEADERS}/simdjson/${header}")
    file(READ "${path}" content)
    string(REPLACE "#ifndef _WIN32" "#if !defined(_WIN32) && !defined(__SWITCH__)" content "${content}")
    file(WRITE "${path}" "${content}")
endforeach()
target_include_directories(simdjson BEFORE PUBLIC "${CT_SIMDJSON_HEADERS}")
set(FASTGLTF_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FASTGLTF_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(FASTGLTF_COMPILE_AS_CPP20 ON CACHE BOOL "" FORCE)
add_subdirectory("${CT_PC_BUILD}/_deps/fastgltf-src" fastgltf EXCLUDE_FROM_ALL)
add_custom_target(switch_libraries DEPENDS Luau.VM Luau.Compiler fastgltf)
# File paths come from the targets, rather than assuming a host archive layout.
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/libraries.json" CONTENT
    "{\n  \"platform\": \"NintendoSwitch\",\n  \"archives\": [\n    \"$<TARGET_FILE:Luau.Compiler>\",\n    \"$<TARGET_FILE:Luau.Ast>\",\n    \"$<TARGET_FILE:Luau.Bytecode>\",\n    \"$<TARGET_FILE:Luau.Common>\",\n    \"$<TARGET_FILE:Luau.VM>\",\n    \"$<TARGET_FILE:fastgltf>\",\n    \"$<TARGET_FILE:simdjson>\"\n  ]\n}\n")
